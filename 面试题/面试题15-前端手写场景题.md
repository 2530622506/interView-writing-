### 防抖 debounce

> 实现一个防抖函数 debounce，支持 immediate 参数（是否立即执行），并提供 cancel 取消功能。

**参考答案：**

**基础版（面试快速版）：**

```js
// 基础防抖：delay 内没有新触发才执行
function debounce(fn, delay) {
  let timer = null
  return function (...args) {
    clearTimeout(timer)
    timer = setTimeout(() => {
      fn.apply(this, args)
    }, delay)
  }
}
```

**完整版（含 immediate + cancel）：**

```js
/**
 * 防抖：连续触发时只执行最后一次
 * @param {Function} fn - 要防抖的函数
 * @param {number} delay - 延迟时间（ms）
 * @param {boolean} immediate - 是否在首次触发时立即执行
 */
function debounce(fn, delay, immediate = false) {
  let timer = null

  const debounced = function (...args) {
    // 如果已有定时器，清除它（重置计时）
    if (timer) clearTimeout(timer)

    if (immediate) {
      // 立即执行模式：第一次触发时立即执行，之后 delay 内不再执行
      const callNow = !timer  // timer 为 null 说明是第一次（或已冷却）
      timer = setTimeout(() => {
        timer = null  // delay 后重置，允许下一次立即执行
      }, delay)
      if (callNow) fn.apply(this, args)
    } else {
      // 普通模式：delay 内没有新触发才执行
      timer = setTimeout(() => {
        fn.apply(this, args)
        timer = null
      }, delay)
    }
  }

  // 取消防抖，清除定时器
  debounced.cancel = function () {
    clearTimeout(timer)
    timer = null
  }

  return debounced
}

// 使用示例
const handleSearch = debounce((keyword) => {
  console.log('搜索:', keyword)
}, 500)

const handleSearchImmediate = debounce((keyword) => {
  console.log('立即搜索:', keyword)
}, 500, true)
```

### 节流 throttle

> 实现一个节流函数 throttle，首次立即执行，最后一次也会触发（结合时间戳版 + 定时器版的优点）。

**参考答案：**

**基础版（时间戳版，首次立即执行，但最后一次不触发）：**

```js
// 基础节流（时间戳版）：固定时间内只执行第一次
function throttle(fn, wait) {
  let lastTime = 0
  return function (...args) {
    const now = Date.now()
    if (now - lastTime >= wait) {
      lastTime = now
      fn.apply(this, args)
    }
  }
}
```

**完整版（时间戳 + 定时器，首次立即执行 + 最后一次也触发）：**

```js
/**
 * 节流：固定时间内只执行一次
 * 综合版：首次立即执行，最后一次也会触发
 * @param {Function} fn - 要节流的函数
 * @param {number} wait - 间隔时间（ms）
 */
function throttle(fn, wait) {
  let lastTime = 0   // 上次执行时间
  let timer = null   // 尾部定时器

  return function (...args) {
    const now = Date.now()
    const remaining = wait - (now - lastTime)  // 距离下次可执行还剩多久

    if (remaining <= 0) {
      // 可以执行：距上次执行已超过 wait
      if (timer) {
        clearTimeout(timer)
        timer = null
      }
      lastTime = now
      fn.apply(this, args)
    } else {
      // 还不到时间：设置尾部定时器，保证最后一次触发能执行
      clearTimeout(timer)
      timer = setTimeout(() => {
        lastTime = Date.now()
        timer = null
        fn.apply(this, args)
      }, remaining)
    }
  }
}

// 使用示例
const handleScroll = throttle(() => {
  console.log('scroll 事件', Date.now())
}, 300)
window.addEventListener('scroll', handleScroll)
```

### 深拷贝 deepClone

> 实现一个完整的深拷贝函数，处理循环引用、Date、RegExp、Map、Set、数组、普通对象。

**参考答案：**

```js
/**
 * 深拷贝
 * @param {any} target - 要拷贝的值
 * @param {WeakMap} map - 用于处理循环引用（WeakMap 允许 GC 回收键）
 */
function deepClone(target, map = new WeakMap()) {
  // 基本类型和函数直接返回
  if (target === null || typeof target !== 'object') return target

  // 处理循环引用：已克隆过的对象直接返回缓存
  if (map.has(target)) return map.get(target)

  // 处理 Date
  if (target instanceof Date) return new Date(target.getTime())

  // 处理 RegExp
  if (target instanceof RegExp) return new RegExp(target.source, target.flags)

  // 处理 Map
  if (target instanceof Map) {
    const clonedMap = new Map()
    map.set(target, clonedMap)  // 先缓存，防止循环引用
    target.forEach((value, key) => {
      clonedMap.set(deepClone(key, map), deepClone(value, map))
    })
    return clonedMap
  }

  // 处理 Set
  if (target instanceof Set) {
    const clonedSet = new Set()
    map.set(target, clonedSet)
    target.forEach(value => {
      clonedSet.add(deepClone(value, map))
    })
    return clonedSet
  }

  // 处理数组和普通对象（保留原型链类型）
  const cloned = Array.isArray(target) ? [] : Object.create(Object.getPrototypeOf(target))
  map.set(target, cloned)  // 先缓存

  // 处理 Symbol 键
  const keys = [...Object.keys(target), ...Object.getOwnPropertySymbols(target)]
  keys.forEach(key => {
    cloned[key] = deepClone(target[key], map)
  })

  return cloned
}

// 使用示例
const obj = { a: 1, b: { c: [1, 2, 3] }, d: new Date(), e: /hello/gi }
obj.self = obj  // 循环引用
const cloned = deepClone(obj)
console.log(cloned.b === obj.b)  // false，深拷贝
console.log(cloned.self === cloned)  // true，循环引用正确处理
```

### 柯里化 curry

> 实现一个 curry 函数，支持任意参数数量，参数不足时返回新函数，参数足够时执行。

**参考答案：**

```js
/**
 * 柯里化：将多参数函数转为接受单参数的函数序列
 * @param {Function} fn - 要柯里化的函数
 */
function curry(fn) {
  return function curried(...args) {
    // 参数足够：直接调用原函数
    if (args.length >= fn.length) {
      return fn.apply(this, args)
    }
    // 参数不够：返回新函数，合并已有参数
    return function (...moreArgs) {
      return curried.apply(this, [...args, ...moreArgs])
    }
  }
}

// 使用示例
const add = (a, b, c) => a + b + c
const curriedAdd = curry(add)

console.log(curriedAdd(1)(2)(3))    // 6
console.log(curriedAdd(1, 2)(3))    // 6
console.log(curriedAdd(1)(2, 3))    // 6
console.log(curriedAdd(1, 2, 3))    // 6
```

### 函数组合 compose / pipe

> 实现 compose（从右到左执行）和 pipe（从左到右执行）。

**参考答案：**

```js
/**
 * compose：从右到左执行函数，前一个函数的输出是后一个函数的输入
 * f(g(h(x))) === compose(f, g, h)(x)
 */
function compose(...fns) {
  return function (x) {
    return fns.reduceRight((acc, fn) => fn(acc), x)
  }
}

/**
 * pipe：从左到右执行函数（compose 的反向版本）
 * f(g(h(x))) === pipe(h, g, f)(x)
 */
function pipe(...fns) {
  return function (x) {
    return fns.reduce((acc, fn) => fn(acc), x)
  }
}

// 使用示例
const double = x => x * 2
const addOne = x => x + 1
const square = x => x * x

const transform = compose(square, addOne, double)  // square(addOne(double(3)))
console.log(transform(3))  // square(addOne(6)) = square(7) = 49

const transform2 = pipe(double, addOne, square)    // square(addOne(double(3)))
console.log(transform2(3))  // 49，和 compose 结果相同（顺序相反）
```

### 深比较 isEqual

> 实现一个深度比较函数 isEqual，处理对象、数组、Date、RegExp 等。

**参考答案：**

```js
/**
 * 深度比较两个值是否相等
 */
function isEqual(a, b) {
  // 严格相等（包括 NaN !== NaN 的特殊处理）
  if (Object.is(a, b)) return true

  // 类型不同直接返回 false
  if (typeof a !== typeof b) return false
  if (a === null || b === null) return false
  if (typeof a !== 'object') return false

  // Date 比较时间戳
  if (a instanceof Date && b instanceof Date) {
    return a.getTime() === b.getTime()
  }

  // RegExp 比较 source 和 flags
  if (a instanceof RegExp && b instanceof RegExp) {
    return a.source === b.source && a.flags === b.flags
  }

  // 类型不一致（如一个是数组一个是对象）
  if (Array.isArray(a) !== Array.isArray(b)) return false

  // 键数量不同
  const keysA = Object.keys(a)
  const keysB = Object.keys(b)
  if (keysA.length !== keysB.length) return false

  // 递归比较每个属性
  return keysA.every(key => isEqual(a[key], b[key]))
}

// 使用示例
console.log(isEqual({ a: 1, b: [2, 3] }, { a: 1, b: [2, 3] }))  // true
console.log(isEqual(new Date('2024-01-01'), new Date('2024-01-01')))  // true
console.log(isEqual([1, [2, 3]], [1, [2, 4]]))  // false
```

### 函数记忆 memoize

> 实现一个 memoize 函数，缓存函数计算结果，相同参数直接返回缓存。

**参考答案：**

```js
/**
 * 函数记忆：缓存计算结果，避免重复计算
 * @param {Function} fn - 要缓存的函数
 * @param {Function} resolver - 自定义缓存 key 生成函数（可选）
 */
function memoize(fn, resolver) {
  const cache = new Map()

  return function (...args) {
    // 生成缓存 key（默认用第一个参数，或自定义序列化）
    const key = resolver ? resolver(...args) : JSON.stringify(args)

    if (cache.has(key)) {
      return cache.get(key)  // 命中缓存
    }

    const result = fn.apply(this, args)
    cache.set(key, result)
    return result
  }
}

// 使用示例：缓存斐波那契计算
const fib = memoize(function (n) {
  if (n <= 1) return n
  return fib(n - 1) + fib(n - 2)
})

console.log(fib(40))  // 快速计算，不会重复递归
```

### 手写 new

> 手写实现 new 操作符的完整过程。

**参考答案：**

```js
/**
 * 手写 new
 * new 的过程：
 * 1. 创建一个空对象
 * 2. 将对象的原型指向构造函数的 prototype
 * 3. 执行构造函数，绑定 this 为新对象
 * 4. 如果构造函数返回的是对象，则返回该对象；否则返回新创建的对象
 */
function myNew(Constructor, ...args) {
  // 1. 创建空对象，原型指向构造函数的 prototype
  const obj = Object.create(Constructor.prototype)

  // 2. 执行构造函数，this 指向新对象
  const result = Constructor.apply(obj, args)

  // 3. 如果构造函数显式返回了一个对象，则以该对象为准；否则返回 obj
  return result instanceof Object ? result : obj
}

// 使用示例
function Person(name, age) {
  this.name = name
  this.age = age
}
Person.prototype.greet = function () {
  return `Hi, I'm ${this.name}`
}

const p = myNew(Person, 'Alice', 25)
console.log(p.name)    // 'Alice'
console.log(p.greet()) // "Hi, I'm Alice"
console.log(p instanceof Person)  // true
```

### 手写 call / apply

> 手写实现 Function.prototype.call 和 apply。

**参考答案：**

```js
/**
 * 手写 call
 * 思路：将函数作为 context 的属性临时挂载，调用后删除
 * 用 Symbol 防止属性名冲突
 */
Function.prototype.myCall = function (context, ...args) {
  // context 为 null/undefined 时指向全局对象
  context = context == null ? globalThis : Object(context)

  // 用 Symbol 作为临时属性名，避免覆盖原有属性
  const key = Symbol('temp')
  context[key] = this  // this 就是调用 myCall 的函数

  const result = context[key](...args)  // 执行函数
  delete context[key]  // 清理临时属性

  return result
}

/**
 * 手写 apply
 * 与 call 的区别：参数以数组形式传入
 */
Function.prototype.myApply = function (context, args = []) {
  context = context == null ? globalThis : Object(context)
  const key = Symbol('temp')
  context[key] = this
  const result = context[key](...args)
  delete context[key]
  return result
}

// 使用示例
function greet(greeting, punctuation) {
  return `${greeting}, ${this.name}${punctuation}`
}
const user = { name: 'Alice' }
console.log(greet.myCall(user, 'Hello', '!'))   // "Hello, Alice!"
console.log(greet.myApply(user, ['Hi', '~']))   // "Hi, Alice~"
```

### 手写 bind

> 手写实现 Function.prototype.bind，处理 new 调用时 this 的修正。

**参考答案：**

```js
/**
 * 手写 bind
 * 关键点：bind 返回的函数被 new 调用时，this 应指向新创建的实例，而不是绑定的 ctx
 */
Function.prototype.myBind = function (ctx, ...preArgs) {
  const fn = this  // 保存原函数引用

  function Bound(...laterArgs) {
    // 判断是否被 new 调用：new 调用时 this 是 Bound 的实例
    // new 调用优先级高于 bind 的 ctx 绑定
    return fn.apply(
      this instanceof Bound ? this : ctx,
      [...preArgs, ...laterArgs]  // 合并预填参数和后续参数
    )
  }

  // 维护原型链：Bound 的实例也是原函数实例
  Bound.prototype = Object.create(fn.prototype)

  return Bound
}

// 使用示例
function Person(name, age) {
  this.name = name
  this.age = age
}

const BoundPerson = Person.myBind({ x: 1 }, 'Alice')
const p = new BoundPerson(25)  // new 调用时 this 不是 { x: 1 }
console.log(p.name)  // 'Alice'
console.log(p instanceof Person)  // true

const boundGreet = function (greeting) {
  return `${greeting}, ${this.name}`
}.myBind({ name: 'Bob' })
console.log(boundGreet('Hello'))  // "Hello, Bob"
```

### 手写 instanceof

```js
/**
 * 手写 instanceof
 * 原理：沿着左侧对象的原型链向上查找，看是否能找到右侧构造函数的 prototype
 */
function myInstanceof(left, right) {
  // 基本类型直接返回 false
  if (typeof left !== 'object' && typeof left !== 'function') return false
  if (left === null) return false

  let proto = Object.getPrototypeOf(left)  // 获取原型

  while (proto !== null) {
    if (proto === right.prototype) return true  // 找到了
    proto = Object.getPrototypeOf(proto)  // 继续向上找
  }

  return false  // 原型链顶端（null）都没找到
}

// 使用示例
console.log(myInstanceof([], Array))    // true
console.log(myInstanceof([], Object))   // true
console.log(myInstanceof({}, Array))    // false
console.log(myInstanceof('str', String)) // false（基本类型）
```

### 并发控制调度器 Scheduler

> 实现一个并发调度器，限制最大并发数为 n，任务完成后自动从队列取下一个任务执行。

**参考答案：**

```js
/**
 * 并发控制调度器（2024-2025 最高频手写题）
 * 核心思路：维护运行中的任务数量，达到上限则排队等待
 */
class Scheduler {
  constructor(max) {
    this.max = max      // 最大并发数
    this.count = 0      // 当前运行中的任务数
    this.queue = []     // 等待队列
  }

  /**
   * 添加任务
   * @param {Function} task - 返回 Promise 的任务函数
   * @returns {Promise} 任务完成时 resolve
   */
  add(task) {
    return new Promise((resolve, reject) => {
      // 将任务和其 resolve/reject 一起推入队列
      this.queue.push({ task, resolve, reject })
      this._run()
    })
  }

  _run() {
    // 有空位且有等待任务，就启动
    while (this.count < this.max && this.queue.length > 0) {
      const { task, resolve, reject } = this.queue.shift()
      this.count++

      task()
        .then(resolve, reject)  // 传递结果
        .finally(() => {
          this.count--   // 释放槽位
          this._run()    // 尝试启动下一个
        })
    }
  }
}

// 使用示例
const scheduler = new Scheduler(2)  // 最多同时运行 2 个任务

const addTask = (delay, name) => {
  scheduler.add(() => new Promise(resolve => {
    setTimeout(() => {
      console.log(`任务 ${name} 完成`)
      resolve(name)
    }, delay)
  }))
}

addTask(1000, 'A')  // 1s 后完成
addTask(500, 'B')   // 0.5s 后完成
addTask(300, 'C')   // A/B 其中之一完成后才开始
addTask(400, 'D')

// 执行顺序：A 和 B 同时开始
// 0.5s: B 完成，C 开始
// 0.8s: C 完成，D 开始
// 1.0s: A 完成
// 1.2s: D 完成
```

```js
function myPromiseAll(promises) {
  return new Promise((resolve, reject) => {
    const results = []
    let resolvedCount = 0
    const total = promises.length

    if (total === 0) return resolve([])

    promises.forEach((p, index) => {

      Promise.resolve(p).then(value => {

        resolvedCount++
        if (resolvedCount === total) resolve(results)  // 全部完成
      }).catch(reject)  // 任一失败立即 reject
    })
  })
```

### 手写 Promise.any

> 实现 Promise.any，第一个成功的 Promise 即返回，全部失败才 reject（AggregateError）。

**参考答案：**

```js
function myPromiseAny(promises) {
  return new Promise((resolve, reject) => {
    const errors = []
    let rejectedCount = 0

    if (promises.length === 0) {
      return reject(new AggregateError([], 'All promises were rejected'))
    }

    promises.forEach((p, index) => {
      Promise.resolve(p)
        .then(resolve)  // 第一个成功直接 resolve
        .catch(err => {
          errors[index] = err
          rejectedCount++
          // 全部失败才 reject
          if (rejectedCount === promises.length) {
            reject(new AggregateError(errors, 'All promises were rejected'))
          }
```

### LazyMan

> 实现 LazyMan 链式调用，支持 sleep 延迟执行。

**参考答案：**

```js
/**
 * LazyMan：链式调用 + 任务队列 + 异步调度
 * LazyMan('Hank').sleep(1).eat('dinner').sleepFirst(2)
 * 输出：等2s -> Hi Hank -> 等1s -> Eat dinner
 */
class LazyManClass {
  constructor(name) {
    this.tasks = []  // 任务队列

    // 默认第一个任务：打招呼
    this.tasks.push(() => {
      console.log(`Hi ${name}`)
      this._next()  // 执行完后触发下一个
    })

    // 用 setTimeout 0 让所有链式调用先注册完，再开始执行
    setTimeout(() => this._next(), 0)
  }

  _next() {
    const task = this.tasks.shift()
    task && task()
  }

  eat(food) {
    this.tasks.push(() => {
      console.log(`Eat ${food}`)
      this._next()
    })
    return this  // 返回 this 支持链式调用
  }

  sleep(seconds) {
    this.tasks.push(() => {
      setTimeout(() => {
        console.log(`等待了 ${seconds} 秒`)
        this._next()
      }, seconds * 1000)
    })
    return this
  }

  sleepFirst(seconds) {
    // 插入到队列最前面
    this.tasks.unshift(() => {
      setTimeout(() => {
        console.log(`等待了 ${seconds} 秒`)
        this._next()
      }, seconds * 1000)
    })
    return this
  }
}

function LazyMan(name) {
  return new LazyManClass(name)
}

// 使用示例

// 输出：Hi Hank -> (等1秒) -> Eat dinner
```

### 手写 Promise 简版

> 手写一个符合 Promise/A+ 规范的简易 Promise，包含状态机、then 链、微任务调度。

**参考答案：**

```js
/**
 * 简易 Promise 实现
 * 三种状态：pending -> fulfilled / rejected（单向不可逆）
 */
class MyPromise {
  constructor(executor) {
    this.state = 'pending'
    this.value = undefined
    this.reason = undefined
    this.onFulfilledCallbacks = []  // 成功回调队列
    this.onRejectedCallbacks = []   // 失败回调队列

    const resolve = (value) => {
      if (this.state !== 'pending') return
      this.state = 'fulfilled'
      this.value = value
      this.onFulfilledCallbacks.forEach(fn => fn(value))
    }

    const reject = (reason) => {
      if (this.state !== 'pending') return
      this.state = 'rejected'
      this.reason = reason
      this.onRejectedCallbacks.forEach(fn => fn(reason))
    }

    try {
      executor(resolve, reject)
    } catch (e) {
      reject(e)
    }
  }

  then(onFulfilled, onRejected) {
    // 参数默认值（值穿透）
    onFulfilled = typeof onFulfilled === 'function' ? onFulfilled : v => v
    onRejected = typeof onRejected === 'function' ? onRejected : e => { throw e }

    // 返回新的 Promise 支持链式调用
    return new MyPromise((resolve, reject) => {
      const handleFulfilled = (value) => {
        // 微任务调度（模拟 Promise 异步执行）
        queueMicrotask(() => {
          try {
            const result = onFulfilled(value)
            // 如果回调返回 Promise，等待其完成
            result instanceof MyPromise ? result.then(resolve, reject) : resolve(result)
          } catch (e) {
            reject(e)
          }
        })
      }

      const handleRejected = (reason) => {
        queueMicrotask(() => {
          try {
            const result = onRejected(reason)
            result instanceof MyPromise ? result.then(resolve, reject) : resolve(result)
          } catch (e) {
            reject(e)
          }
        })
      }

      if (this.state === 'fulfilled') handleFulfilled(this.value)
      else if (this.state === 'rejected') handleRejected(this.reason)
      else {
        // pending 状态：将回调存入队列，等待 resolve/reject 触发
        this.onFulfilledCallbacks.push(handleFulfilled)
        this.onRejectedCallbacks.push(handleRejected)
      }
    })
  }

  catch(onRejected) {
    return this.then(null, onRejected)
  }

  static resolve(value) {
    return value instanceof MyPromise ? value : new MyPromise(resolve => resolve(value))
  }

  static reject(reason) {
    return new MyPromise((_, reject) => reject(reason))
  }
}
```

```js
function flattenRecursive(arr) {
  return arr.reduce((acc, item) => {
    return Array.isArray(item)
      ? acc.concat(flattenRecursive(item))  // 递归处理嵌套数组
      : acc.concat(item)
  }, [])
}

// 方案二：支持指定深度（depth）
function flatten(arr, depth = Infinity) {
  if (depth === 0) return arr.slice()  // 深度为 0 不展开

  return arr.reduce((acc, item) => {
    if (Array.isArray(item) && depth > 0) {
      acc.push(...flatten(item, depth - 1))  // 深度减 1 递归
    } else {
      acc.push(item)
    }
    return acc
  }, [])
}

// 方案三：栈迭代（非递归，避免栈溢出）
function flattenStack(arr) {
  const stack = [...arr]  // 将数组转为栈
  const result = []

  while (stack.length) {
    const item = stack.pop()  // 从末尾取出
    if (Array.isArray(item)) {
      stack.push(...item)  // 展开后重新压栈
    } else {
      result.unshift(item)  // 非数组放入结果（用 unshift 保持顺序）
    }
  }
  return result
}
```

```js
/**
 * 数组转树（Map O(n) 最优解）
 * 对比递归解法（O(n²)），Map 解法只需遍历两次
 */
function arrayToTree(arr) {
  // 第一步：用 Map 存储所有节点（id -> 节点），并初始化 children
  const map = new Map()
  arr.forEach(item => {
    map.set(item.id, { ...item, children: [] })

  const roots = []

  // 第二步：遍历建立父子关系

    const node = map.get(item.id)
    if (item.parentId === null) {
      roots.push(node)  // 根节点
    } else {
      const parent = map.get(item.parentId)
      if (parent) parent.children.push(node)  // 挂载到父节点
    }
  })

  return roots
}
```

```js
// DFS（深度优先，递归）
function treeToArrayDFS(tree) {
  const result = []

  function dfs(nodes) {
    nodes.forEach(node => {
      const { children, ...rest } = node
      result.push(rest)
      if (children && children.length) dfs(children)  // 递归处理子节点
    })
  }

  dfs(tree)
  return result
}

// BFS（广度优先，队列）
function treeToArrayBFS(tree) {
  const result = []
  const queue = [...tree]  // 初始化队列

  while (queue.length) {
    const node = queue.shift()  // 取出队头
    const { children, ...rest } = node
    result.push(rest)
    if (children && children.length) queue.push(...children)  // 子节点入队
  }

  return result
}
```

### LRU 缓存

> 实现 LRU 缓存，get 和 put 操作都是 O(1)。

**参考答案：**

```js
/**
 * LRU（最近最少使用）缓存
 * 利用 Map 的插入顺序特性：最早插入的 key 在迭代器最前面
 * get/put 时删除再重新插入，相当于移到"最近使用"位置
 */
class LRUCache {
  constructor(capacity) {
    this.capacity = capacity
    this.map = new Map()  // Map 保证插入顺序
  }

  get(key) {
    if (!this.map.has(key)) return -1

    const value = this.map.get(key)
    // 移到末尾（标记为最近使用）
    this.map.delete(key)
    this.map.set(key, value)
    return value
  }

  put(key, value) {
    if (this.map.has(key)) {
      this.map.delete(key)  // 先删除，后面重新插入（移到末尾）
    } else if (this.map.size >= this.capacity) {
      // 超出容量：删除最久未使用的（Map 迭代器的第一个）
      const oldestKey = this.map.keys().next().value
      this.map.delete(oldestKey)
    }
    this.map.set(key, value)
  }
}

// 使用示例
const cache = new LRUCache(2)
cache.put(1, 1)
cache.put(2, 2)
console.log(cache.get(1))  // 1（访问后 1 成为最近使用）
cache.put(3, 3)            // 2 是最久未使用，被淘汰
console.log(cache.get(2))  // -1（已被淘汰）
console.log(cache.get(3))  // 3
```

### 实现 JSON.stringify

> 手写实现 JSON.stringify，处理各种类型和循环引用检测。

**参考答案：**

```js
/**
 * 手写 JSON.stringify
 * 处理：string/number/boolean/null/undefined/Date/Array/Object/函数/Symbol/循环引用
 */
function myJSONStringify(value, seen = new Set()) {
  // 特殊值处理
  if (value === undefined || typeof value === 'function' || typeof value === 'symbol') {
    return undefined  // 这些类型在对象中会被忽略，在数组中变 null
  }
  if (value === null) return 'null'
  if (typeof value === 'boolean') return String(value)
  if (typeof value === 'number') {
    // NaN 和 Infinity 序列化为 null
    return isFinite(value) ? String(value) : 'null'
  }
  if (typeof value === 'string') {
    return `"${value.replace(/"/g, '\\"')}"` // 转义双引号
  }
  if (value instanceof Date) {
    return `"${value.toISOString()}"`
  }

  // 检测循环引用
  if (seen.has(value)) throw new TypeError('循环引用')
  seen.add(value)

  if (Array.isArray(value)) {
    const items = value.map(item => {
      const result = myJSONStringify(item, seen)
      return result === undefined ? 'null' : result  // 数组中的 undefined 变 null
    })
    seen.delete(value)
    return `[${items.join(',')}]`
  }

  // 普通对象
  const entries = Object.keys(value)
    .map(key => {
      const val = myJSONStringify(value[key], seen)
      if (val === undefined) return null  // 忽略 undefined、函数、Symbol 值
      return `"${key}":${val}`
    })
    .filter(Boolean)

  seen.delete(value)
  return `{${entries.join(',')}}`
}

// 使用示例
console.log(myJSONStringify({ a: 1, b: [2, 3], c: null, d: undefined }))
// '{"a":1,"b":[2,3],"c":null}'（d:undefined 被忽略）
```

### 虚拟列表

> 手写一个简版虚拟列表，只渲染可视区域内的条目，解决大列表性能问题。

**参考答案：**

```js
/**
 * 虚拟列表（定高版）
 * 核心思路：
 * 1. 容器高度 = 总数据量 × 单项高度（撑开滚动条）
 * 2. 只渲染可视区域内的条目（startIndex ~ endIndex）
 * 3. 用 transform translateY 将渲染的条目定位到正确位置
 */
class VirtualList {
  constructor({ container, itemHeight, data, renderItem }) {
    this.container = container
    this.itemHeight = itemHeight
    this.data = data
    this.renderItem = renderItem
    this.bufferSize = 3  // 上下各多渲染几条，防止快速滚动闪白

    this._init()
  }

  _init() {
    // 外层容器：设置溢出滚动
    this.container.style.overflow = 'auto'
    this.container.style.position = 'relative'

    // 撑开容器：高度 = 总数 × 单项高度
    this.phantom = document.createElement('div')
    this.phantom.style.height = `${this.data.length * this.itemHeight}px`
    this.phantom.style.position = 'absolute'
    this.phantom.style.width = '100%'
    this.container.appendChild(this.phantom)

    // 实际渲染区域
    this.listEl = document.createElement('div')
    this.listEl.style.position = 'absolute'
    this.listEl.style.width = '100%'
    this.container.appendChild(this.listEl)

    this.container.addEventListener('scroll', () => this._update())
    this._update()
  }

  _update() {
    const scrollTop = this.container.scrollTop
    const visibleHeight = this.container.clientHeight

    // 计算可视区域的起止索引（加缓冲区）
    const startIndex = Math.max(0, Math.floor(scrollTop / this.itemHeight) - this.bufferSize)
    const endIndex = Math.min(
      this.data.length - 1,
      Math.floor((scrollTop + visibleHeight) / this.itemHeight) + this.bufferSize
    )

    // 只渲染 startIndex ~ endIndex 的条目
    this.listEl.innerHTML = ''
    this.listEl.style.transform = `translateY(${startIndex * this.itemHeight}px)`

    for (let i = startIndex; i <= endIndex; i++) {
      const item = document.createElement('div')
      item.style.height = `${this.itemHeight}px`
      item.innerHTML = this.renderItem(this.data[i], i)
      this.listEl.appendChild(item)
    }
  }
}
```

### 发布订阅 EventEmitter

> 手写完整 EventEmitter，支持 on / off / emit / once。

**参考答案：**

```js
class EventEmitter {
  constructor() {
    this._events = {}  // 事件名 -> 回调函数数组
  }

  // 注册事件监听
  on(event, fn) {
    if (!this._events[event]) this._events[event] = []
    this._events[event].push(fn)
    return this  // 返回 this 支持链式调用
  }

  // 取消事件监听
  off(event, fn) {
    if (!this._events[event]) return this
    this._events[event] = this._events[event].filter(
      f => f !== fn && f._origin !== fn  // _origin 用于匹配 once 包装的函数
    )
    return this
  }

  // 触发事件
  emit(event, ...args) {
    const fns = this._events[event]
    if (!fns || fns.length === 0) return this
    // 复制一份再执行，避免 once 在执行中修改数组导致问题
    fns.slice().forEach(fn => fn.apply(this, args))
    return this
  }

  // 只触发一次的监听
  once(event, fn) {
    const wrapper = (...args) => {
      fn.apply(this, args)
      this.off(event, wrapper)  // 执行后立即取消
    }
    wrapper._origin = fn  // 保存原函数引用，让 off 能通过原函数取消
    return this.on(event, wrapper)
  }

  // 清除所有监听（或指定事件的所有监听）
  removeAllListeners(event) {
    if (event) delete this._events[event]
    else this._events = {}
    return this
  }
}

// 使用示例
const emitter = new EventEmitter()
const handler = (msg) => console.log('收到:', msg)

emitter.on('message', handler)
emitter.once('connect', () => console.log('连接成功（只触发一次）'))

emitter.emit('message', 'Hello')  // 收到: Hello
emitter.emit('connect')            // 连接成功（只触发一次）
emitter.emit('connect')            // 不再触发

emitter.off('message', handler)
emitter.emit('message', 'World')   // 不再触发
```

### 图片懒加载

> 使用 IntersectionObserver 实现图片懒加载，图片进入视口时才加载真实 src。

**参考答案：**

```js
/**
 * 图片懒加载（IntersectionObserver 版，比滚动监听性能更好）
 * 使用方法：<img data-src="real-url.jpg" src="placeholder.jpg" class="lazy" />
 */
function lazyLoadImages(selector = '.lazy') {
  const images = document.querySelectorAll(selector)

  const observer = new IntersectionObserver((entries, obs) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        // 图片进入视口
        const img = entry.target
        img.src = img.dataset.src   // 替换为真实 src
        img.classList.remove('lazy')
        obs.unobserve(img)          // 加载后取消观察，避免重复触发
      }
    })
  }, {
    rootMargin: '100px',  // 提前 100px 开始加载
    threshold: 0.1        // 10% 进入视口就触发
  })

  images.forEach(img => observer.observe(img))
}

// 页面加载后调用
document.addEventListener('DOMContentLoaded', () => lazyLoadImages())
```

### URL 参数解析

> 实现 parseQueryString，将 URL query 字符串解析为对象。

**参考答案：**

```js
/**
 * 方案一：手写正则版
 * 处理 ?a=1&b=2&c=hello%20world -> { a: '1', b: '2', c: 'hello world' }
 */
function parseQueryString(url) {
  const result = {}
  const queryStr = url.split('?')[1]
  if (!queryStr) return result

  queryStr.split('&').forEach(pair => {
    const [key, value] = pair.split('=')
    if (key) {
      // decodeURIComponent 处理编码字符
      result[decodeURIComponent(key)] = decodeURIComponent(value || '')
    }
  })

  return result
}

/**
 * 方案二：URLSearchParams API（更简洁，现代浏览器推荐）
 */
function parseQueryStringAPI(url) {
  const params = new URLSearchParams(url.split('?')[1] || '')
  const result = {}
  params.forEach((value, key) => {
    // 处理同名参数（转为数组）
    if (result[key] !== undefined) {
      result[key] = [].concat(result[key], value)
    } else {
      result[key] = value
    }
  })
  return result
}

// 使用示例
const url = 'https://example.com/search?name=Alice&age=25&tags=js&tags=ts'
console.log(parseQueryString(url))
// { name: 'Alice', age: '25', tags: 'ts' }（同名取最后一个）
console.log(parseQueryStringAPI(url))
// { name: 'Alice', age: '25', tags: ['js', 'ts'] }（同名转数组）
```

### 拖拽

> 实现一个可拖拽的元素，使用 mousedown/mousemove/mouseup 事件链。

**参考答案：**

```js
/**
 * 拖拽实现
 * 关键点：
 * 1. mousedown 记录起始偏移量
 * 2. mousemove 更新位置（要在 document 上监听，防止鼠标移出元素时丢失）
 * 3. mouseup 解绑事件
 */
function makeDraggable(element) {
  let isDragging = false
  let offsetX = 0  // 鼠标在元素内的偏移
  let offsetY = 0

  // 确保元素可定位
  element.style.position = 'absolute'
  element.style.cursor = 'grab'

  element.addEventListener('mousedown', (e) => {
    isDragging = true
    // 记录鼠标在元素内的相对位置
    offsetX = e.clientX - element.getBoundingClientRect().left
    offsetY = e.clientY - element.getBoundingClientRect().top
    element.style.cursor = 'grabbing'
    e.preventDefault()  // 防止选中文字
  })

  // 在 document 上监听 mousemove，防止快速移动时鼠标超出元素
  document.addEventListener('mousemove', (e) => {
    if (!isDragging) return

    // 计算新位置
    const x = e.clientX - offsetX
    const y = e.clientY - offsetY

    // 限制在视口内（可选）
    const maxX = window.innerWidth - element.offsetWidth
    const maxY = window.innerHeight - element.offsetHeight

    element.style.left = `${Math.max(0, Math.min(x, maxX))}px`
    element.style.top = `${Math.max(0, Math.min(y, maxY))}px`
  })

  document.addEventListener('mouseup', () => {
    isDragging = false
    element.style.cursor = 'grab'
  })
}
```
